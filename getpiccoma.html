getpiccoma

getpiccoma

Retrieve and save images from manga distribution sites using Piccoma.

Pages are served sliced into a grid of 50px tiles and shuffled; getpiccoma replays the viewer's own shuffle to put them back together before saving.

Valid URL Formats

  • piccoma.com/web/viewer/<product-id>/<episode-id> -- one episode
    • e.g. https://piccoma.com/web/viewer/8195/1185884
  • piccoma.com/web/product/<product-id>/episodes -- the whole series, from its first entry
    • e.g. https://piccoma.com/web/product/8195/episodes

Only Piccoma Japan is supported. Piccoma France (piccoma.com/fr/) runs a different site and is not handled.

# mise via github release
mise use -g github:eggplants/getpiccoma

# mise via pipx
mise use -g pipx:getpiccoma

pipx install getpiccoma

pip install getpiccoma

CLI

$ pget https://piccoma.com/web/viewer/8195/1185884
get: https://piccoma.com/web/viewer/8195/1185884
  Downloading... ━━━━━━━━━━━━ 100% ( 14/14 pages ) remain: 0:00:00 spent: 0:00:09
saved: <series>/<episode>
done.

-b follows every next episode, and stops on the first one the account may not read. A product URL implies it:

pget https://piccoma.com/web/product/8195/episodes -d out

Log in with -u to reach episodes the account has bought or unlocked. The password is prompted for when -p is left out:

pget https://piccoma.com/web/viewer/8195/1185884 -u you@example.com
Option Description
-b, --bulk follow every next episode
-d DIR, --savedir DIR directory to save into
-f, --first download only the first page
-o, --overwrite download again if it exists
-m, --metadata save episode metadata as json
-u MAIL, --username MAIL email address to log in with
-p PW, --password PW password
-t, --volumes read a product url's volume list, not episodes
-q, --quiet disable console output

Library

from getpiccoma import Piccoma

piccoma = Piccoma()
next_url, save_dir, saved = piccoma.get(
    "https://piccoma.com/web/viewer/8195/1185884",
    save_path="out",
)

episode_info reads a viewer page without downloading anything, and entries lists a series:

episode = piccoma.episode_info("https://piccoma.com/web/viewer/8195/1185884")
print(episode.series_title, episode.episode_title, len(episode.pages))

for entry in piccoma.entries(episode.product_id):
    print(entry.title, entry.is_readable, entry.url)

descramble and parse_seed are exposed on their own, for pages fetched some other way:

from PIL import Image
from getpiccoma import descramble, parse_seed

seed = parse_seed(image_url)
page = descramble(Image.open(path), seed) if seed else Image.open(path)

491    def login(self, email: str, password: str) -> None:
492        """Sign in, so episodes the account may read become readable.
493
494        The sign-in form is a plain Django one: a CSRF token off the page goes
495        back with the credentials, and the session cookie lands on the shared
496        session. This grants nothing the account does not already own.
497
498        Args:
499            email: The address the account uses.
500            password: The account's password.
501
502        Raises:
503            LoginError: Piccoma refused the credentials.
504        """
505        form = self._session.get(LOGIN_URL, headers=HEADERS, timeout=30)
506        form.raise_for_status()
507        token = BeautifulSoup(form.content, "html.parser").find("input", attrs={"name": "csrfmiddlewaretoken"})
508        if not isinstance(token, Tag):
509            msg = f"{LOGIN_URL} carries no sign-in form."
510            raise LoginError(msg)
511
512        res = self._session.post(
513            LOGIN_URL,
514            data={
515                "csrfmiddlewaretoken": str(token.attrs.get("value", "")),
516                "next_url": "/web/",
517                "email": email,
518                "password": password,
519            },
520            headers={**HEADERS, "Origin": BASE_URL, "Referer": LOGIN_URL},
521            timeout=30,
522        )
523        res.raise_for_status()
524
525        if not self._login_status():
526            msg = f"piccoma.com refused the credentials for {email!r}."
527            raise LoginError(msg)
528        self._logged_in = True

530    def episode_info(self, url: str) -> Episode:
531        """Read the page list and the titles off a viewer page.
532
533        Args:
534            url: The viewer URL.
535
536        Returns:
537            The parsed episode. `pages` is empty when it is not readable.
538
539        Raises:
540            NotAPiccomaPageError: The page carries no viewer.
541        """
542        res = self._session.get(url, headers=HEADERS, timeout=30)
543        res.raise_for_status()
544        if urlparse(res.url).path.startswith(_SIGNIN_PREFIX):
545            return self._locked_episode(url)
546        html = res.text
547
548        block = _PDATA.search(html)
549        if block is None:
550            msg = f"no '_pdata_' on {url}; is it a Piccoma viewer page?"
551            raise NotAPiccomaPageError(msg)
552        body = block.group(1)
553        fields = {match["key"]: _literal(match["value"]) for match in _PDATA_FIELD.finditer(body)}
554
555        product_id = str(fields.get("product_id") or "")
556        episode_id = str(fields.get("episode_id") or "")
557        episode_type: EpisodeType = "V" if fields.get("eType") == "V" else "E"
558        episode_title = str(fields.get("title") or "").strip() or episode_id
559
560        return Episode(
561            url=url,
562            product_id=product_id,
563            episode_id=episode_id,
564            series_title=_title_part(BeautifulSoup(html, "html.parser"), 1) or product_id,
565            episode_title=episode_title,
566            episode_type=episode_type,
567            scrambled=bool(fields.get("isScrambled")),
568            pages=tuple(_pages(body)),
569            next_url=self._next_url(product_id, episode_id, episode_type),
570        )

572    def entries(self, product_id: str, episode_type: EpisodeType = "E") -> list[Entry]:
573        """List a series' episodes or volumes, in reading order.
574
575        The result is cached, so walking a whole series costs one request for
576        the list however many episodes are downloaded off it.
577
578        Args:
579            product_id: The series' id.
580            episode_type: `"E"` for the episode list, `"V"` for the volume one.
581
582        Returns:
583            The rows of the list. Empty when the product has none of that kind.
584        """
585        cached = self._lists.get((product_id, episode_type))
586        if cached is not None:
587            return cached
588
589        url = f"{BASE_URL}/web/product/{product_id}/episodes?etype={episode_type}"
590        res = self._session.get(url, headers=HEADERS, timeout=30)
591        res.raise_for_status()
592        soup = BeautifulSoup(res.content, "html.parser")
593        # The product page names the series; the viewer page of a locked
594        # episode does not, so it is worth keeping hold of.
595        self._series_titles.setdefault(product_id, _title_part(soup, 0))
596        listing = soup.find(id="js_volumeList" if episode_type == "V" else "js_episodeList")
597
598        entries: list[Entry] = []
599        if isinstance(listing, Tag):
600            entries = [
601                entry
602                for item in listing.find_all("li", recursive=False)
603                if isinstance(item, Tag) and (entry := _entry(item, product_id, episode_type)) is not None
604            ]
605        self._lists[product_id, episode_type] = entries
606        return entries

621    def _locked_episode(self, url: str) -> Episode:
622        """Describe an episode Piccoma would not open, from its series' list."""
623        product_id, episode_id = self.parse_uri(url)
624        episode_id = episode_id or ""
625        entry = None
626        for episode_type in ("E", "V"):
627            entry = next((row for row in self.entries(product_id, episode_type) if row.id == episode_id), None)
628            if entry is not None:
629                break
630        return Episode(
631            url=url,
632            product_id=product_id,
633            episode_id=episode_id,
634            series_title=self.series_title(product_id),
635            episode_title=entry.title if entry else episode_id,
636            episode_type="E",
637            scrambled=False,
638            next_url=self._next_url(product_id, episode_id, "E"),
639        )
640
641    def _next_url(self, product_id: str, episode_id: str, episode_type: EpisodeType) -> str | None:
642        """Find the episode that follows `episode_id` in its series' list."""
643        if not product_id or not episode_id:
644            return None
645        entries = self.entries(product_id, episode_type)
646        ids = [entry.id for entry in entries]
647        try:
648            position = ids.index(episode_id)
649        except ValueError:
650            return None
651        return entries[position + 1].url if position + 1 < len(entries) else None

653    def _login_status(self) -> bool:
654        """Ask any page whether the session is signed in."""
655        res = self._session.get(LOGIN_URL, headers=HEADERS, timeout=30)

660    def _save_pages(
661        self,
662        episode: Episode,
663        save_dir: Path,
664        *,
665        only_first: bool = False,
666        print_log: bool = False,
667    ) -> None:
668        wanted = episode.pages[:1] if only_first else episode.pages
669        width = len(str(len(wanted)))
670        progress = Progress(
671            SpinnerColumn(),
672            TextColumn("{task.description}"),
673            BarColumn(),

677            TextColumn("pages )"),

691    def _image(self, page: Page, *, scrambled: bool) -> Image.Image:
692        res = self._session.get(page["url"], headers=HEADERS, timeout=60)

334    """Put a shuffled page back together.
335
336    Args:
337        image: The page exactly as the CDN serves it.
338        seed: The seed string, from `parse_seed`.
339        tile_size: The side of a full tile.
340
341    Returns:
342        A new image with the tiles back where they belong.
343    """
344    out = image.copy()
345    for corners in _tile_groups(*image.size, tile_size):
346        tile_width = min(tile_size, image.width - corners[0][0])
347        tile_height = min(tile_size, image.height - corners[0][1])
348        group_x, group_y = corners[0]
349        # A group's own grid is as wide as the run of tiles sharing its first row.

355            out.paste(image.crop((x, y, x + tile_width, y + tile_height)), (dest_x, dest_y))
356    return out

272def parse_seed(image_url: str) -> str | None:
273    """Work out the seed a page image was shuffled with.
274
275    The CDN path carries a per-episode checksum, which the viewer rotates right
276    by every non-zero digit of the URL's `expires` stamp and then perturbs.
277    Unscrambled pages are served under a lowercase checksum instead.
278
279    Args:
280        image_url: The page's image URL, `expires` query and all.
281
282    Returns:
283        The seed string, or None when the image is not scrambled.
284
285    Raises:
286        PiccomaError: The URL carries no checksum or no `expires` stamp.
287    """
288    parsed = urlparse(image_url)
289    segments = [segment for segment in parsed.path.split("/") if segment]
290    if len(segments) < 2:  # noqa: PLR2004 (a checksum and a file name)
291        msg = f"{image_url!r} carries no checksum path segment."
292        raise PiccomaError(msg)
293    checksum = segments[-2]
294
295    expires = parse_qs(parsed.query).get("expires")
296    if not expires:
297        msg = f"{image_url!r} carries no 'expires' stamp to rotate the checksum by."
298        raise PiccomaError(msg)
299
300    for digit in expires[0]:
301        shift = int(digit) if digit.isdigit() else 0
302        if shift:
303            checksum = checksum[-shift:] + checksum[:-shift]
304
305    return _seed_key(checksum) if checksum.isupper() else None

def seedrandom(seed: str) -> Callable[[], float]: 
196def seedrandom(seed: str) -> Callable[[], float]:
197    """Build the seeded PRNG the viewer shuffles its tiles with.
198
199    Args:
200        seed: The seed string, as `parse_seed` returns it.
201
202    Returns:
203        A callable handing out floats in `[0, 1)`.
204
205    Raises:
206        PiccomaError: The seed is empty, which keys nothing.
207    """
208    if not seed:
209        msg = "cannot seed the shuffle with an empty string."
210        raise PiccomaError(msg)
211    arc4 = _ARC4(_mixkey(seed))
212
213    def prng() -> float:
214        numerator: float = arc4.generate(_PRNG_CHUNKS)
215        denominator: float = _PRNG_STARTDENOM
216        extra = 0
217        # Pull more bytes until the fraction carries a full mantissa of them,
218        # then halve it back under the range a double represents exactly.
219        while numerator < _PRNG_SIGNIFICANCE:
220            numerator = (numerator + extra) * _ARC4_WIDTH
221            denominator *= _ARC4_WIDTH
222            extra = arc4.generate(1)
223        while numerator >= _PRNG_OVERFLOW:
224            numerator /= 2
225            denominator /= 2
226            extra >>= 1
227        return (numerator + extra) / denominator
228
229    return prng
